\begin{frame}{Model}
    \[
    y_{it} = \alpha_i + \delta_t + \beta x_{it} + \varepsilon_{it}
    \]
    \begin{itemize}
        \item $y_{it}$: outcome of unit $i$ in period $t$
        \item $\alpha_i$, $\delta_t$: unit and period fixed effects
        \item Identification: $\mathrm{E}(\varepsilon_{it}\mid x_{it},\alpha_i,\delta_t)=0$
    \end{itemize}
\end{frame}

    \begin{itemize}
        \item The decision quantity and its definition in model parameters
        \item The estimate that maps into it
    \end{itemize}
\end{frame}
