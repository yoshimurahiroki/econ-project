%% BEGIN BODY
% Each \slot names what the slide carries; the sequence follows a research talk of the same type
% (exemplars.py find --kind slide) and the content comes from the worksheet. One claim per slide,
% with its exhibit and a one-line reading; delete frames the talk does not use.
\begin{frame}{\slot{Question and stake}}
    \begin{itemize}
        \item \slot{The decision or fact that makes the question matter, with its magnitude}
        \item \slot{The question: who chooses what, and the change under study}
    \end{itemize}
\end{frame}

\begin{frame}{\slot{Answer}}
    \begin{itemize}
        \item \slot{Main estimate with interval, unit, population and period}
        \item \slot{The decision it implies and the condition under which it switches}
    \end{itemize}
\end{frame}

\begin{frame}{\slot{Mechanism}}
    \begin{itemize}
        \item \slot{Competing mechanisms and their predictions for the same change}
        \item \slot{The observation that separates them}
    \end{itemize}
\end{frame}

\begin{frame}{\slot{Setting and data}}
    \begin{itemize}
        \item \slot{The rule or shock, its timing and the agents it reaches}
        \item \slot{Sources, unit of observation, sample and period}
    \end{itemize}
\end{frame}

\begin{frame}{\slot{Identifying comparison}}
    \begin{itemize}
        \item \slot{Estimand, the variation that identifies it and the comparison group}
        \item \slot{The identifying assumption and the method from its method card}
    \end{itemize}
\end{frame}

\begin{frame}{\slot{Main result}}
    \centering
    \IfFileExists{figures/main.pdf}{\includegraphics[height=0.66\textheight]{figures/main.pdf}}{\fbox{\parbox[c][0.55\textheight][c]{0.8\textwidth}{\centering figures/main.pdf}}}

    \raggedright
    \slot{One-line reading: the estimate and the prediction it supports}
\end{frame}

\begin{frame}{\slot{Magnitude and decision}}
    \begin{itemize}
        \item \slot{The estimate converted to the decision's unit: per yen, per worker, elasticity or welfare}
        \item \slot{The decision and its switching value}
    \end{itemize}
\end{frame}

\begin{frame}{\slot{Mechanism evidence}}
    \begin{itemize}
        \item \slot{The test of the separating prediction and its result}
    \end{itemize}
\end{frame}

\begin{frame}{\slot{Conclusion}}
    \begin{itemize}
        \item \slot{The answer, its magnitude and the decision it informs}
    \end{itemize}
\end{frame}
